def _unit(tok):
    tok = ALIAS.get(tok, tok)
    if tok in UNITS: return UNITS[tok]
    for p in sorted(PREFIX, key=len, reverse=True):
        if tok.startswith(p) and tok[len(p):] in PREFIXABLE and len(tok) > len(p):
            f, d = UNITS[tok[len(p):]]; return (f * PREFIX[p], d)
    raise ValueError("Unknown unit “" + tok + "”")

def parse_units(s):
    s = s.strip().replace("·", "*").replace("⁻¹", "^-1").replace("²", "^2").replace("³", "^3").replace("μ", "u").replace("Ω", "ohm").replace("°", "deg")
    if s in ALIAS and "/" in ALIAS[s]: s = ALIAS[s]
    num, _, den = s.partition("/")
    den = den.strip("() ")
    fac, dims = 1.0, [0] * 6
    for part, sign in ((num, 1), (den, -1)):
        for tok in re.split(r"[\s*]+", part.strip("() ")):
            if not tok: continue
            m = re.match(r"^([^\^]+?)(?:\^\(?(-?\d+)\)?)?$", tok)
            if not m: raise ValueError("Cannot read unit “" + tok + "”")
            e = int(m.group(2) or 1) * sign; f, d = _unit(m.group(1))
            fac *= f ** e; dims = [x + y * e for x, y in zip(dims, d)]
    return fac, tuple(dims)

def _temp(v, a, b):
    k = {"K": v, "C": v + 273.15, "F": (v - 32) * 5 / 9 + 273.15}[a]
    return {"K": k, "C": k - 273.15, "F": (k - 273.15) * 9 / 5 + 32}[b]

def op_convert(v, a, b, sf):
    val = float(P(v).evalf()); sf = int(sf)
    if a.strip() in TEMPS and b.strip() in TEMPS:
        r = _temp(val, TEMPS[a.strip()], TEMPS[b.strip()])
        return _res("", "Converted", _fmt(val, sf) + r"\ \mathrm{" + a.strip() + r"} = " + _fmt(r, sf) + r"\ \mathrm{" + b.strip() + "}", [], "", _plain(r, sf) + " " + b)
    fa, da = parse_units(a); fb, db = parse_units(b)
    if da != db: raise ValueError("Those units measure different things, so they cannot be converted")
    r = val * fa / fb
    rows = [["Factor", _fmt(fa / fb, 6)], ["In SI base units", _fmt(val * fa, sf)]]
    tx = lambda u: r"\mathrm{" + u.strip().replace("^", "^{").replace(" ", r"\,") + "}" if "^" not in u else r"\mathrm{" + re.sub(r"\^(-?\d+)", r"^{\1}", u.strip()).replace(" ", r"\,") + "}"
    return _res("", "Converted", _fmt(val, sf) + r"\ " + tx(a) + " = " + _fmt(r, sf) + r"\ " + tx(b), rows, "Answers to " + str(sf) + " significant figures.", _plain(r, sf) + " " + b)

# ---------------- SUVAT ----------------
def op_suvat(vals, sf):
    s_, u_, v_, a_, t_ = sp.symbols("s u v a t", real=True)
    names = {"s": s_, "u": u_, "v": v_, "a": a_, "t": t_}
    known = {k: sp.nsimplify(P(x), rational=True) for k, x in vals.items() if str(x).strip() != ""}
    if len(known) != 3: raise ValueError("Fill exactly three of s, u, v, a, t and leave the other two blank")
    unknown = [names[k] for k in names if k not in known]
    eqs = [v_ - (u_ + a_ * t_), s_ - (u_ + v_) / 2 * t_, s_ - (u_ * t_ + a_ * t_**2 / 2), v_**2 - (u_**2 + 2 * a_ * s_), s_ - (v_ * t_ - a_ * t_**2 / 2)]
    labels = ["v = u + at", "s = ½(u + v)t", "s = ut + ½at²", "v² = u² + 2as", "s = vt − ½at²"]
    sub = {names[k]: x for k, x in known.items()}
    used = []
    for i in range(5):
        for j in range(i + 1, 5):
            e1, e2 = eqs[i].subs(sub), eqs[j].subs(sub)
            if unknown and not ({*unknown} & e1.free_symbols) and not ({*unknown} & e2.free_symbols): continue
            sol = sp.solve([e1, e2], unknown, dict=True)
            sol = [d for d in sol if len(d) == 2 and all(x.is_real for x in d.values())]
            if sol: used = (i, j); break
        if used: break
    if not used: raise ValueError("No real solution for these values")
    good = [d for d in sol if not (t_ in d and d[t_] < 0)]
    good = good or sol
    rows = []
    for k in "suvat":
        if k in known: rows.append([k, _fmt(float(known[k]), sf) + " (given)"])
        else: rows.append([k, " or ".join(_fmt(float(d[names[k]]), sf) for d in good)])
    main = ", ".join(r"%s = %s" % (k, _fmt(float(good[0][names[k]]), sf)) for k in "suvat" if k not in known)
    note = "Equations used: " + labels[used[0]] + " and " + labels[used[1]] + "." + (" Two valid solutions are listed." if len(good) > 1 else "")
    return _res("", "Solved", main, rows, note, main)

BYKEY = {f[0]: f for f in FORMULAS}

def formula_list(): return json.dumps([{"key": f[0], "group": f[1], "name": f[2], "eq": f[3], "vars": [[k, v[0], v[1], v[2] or "", v[3]] for k, v in f[4].items()]} for f in FORMULAS])

def op_formula(key, vals, sf):
    f = BYKEY[key]; sf = int(sf); spec = f[4]
    syms = {k: sp.Symbol(k, real=True) for k in spec}
    lhs, rhs = f[3].split("=", 1)
    eq = sp.sympify(lhs, locals=syms) - sp.sympify(rhs, locals=syms)
    blanks = [k for k in spec if str(vals.get(k, "")).strip() == ""]
    if len(blanks) != 1: raise ValueError("Leave exactly one box empty: that is the quantity to find")
    unk = blanks[0]
    num = {syms[k]: float(P(str(vals[k])).evalf()) for k in spec if k != unk}
    sols = sp.solve(eq, syms[unk])
    if not sols: raise ValueError("This equation has no solution for the values given")
    out = []
    for s_ in sols:
        v = complex(s_.subs(num).evalf())
        if abs(v.imag) < 1e-12: out.append(v.real)
    if not out: raise ValueError("No real value fits these numbers")
    if unk in ("theta", "theta1", "theta2") and key in ("work", "snell"):   # these formulas take degrees
        pass
    lab, unit, _, tx = spec[unk]
    main = tx + " = " + " \\ \\text{or}\\ ".join(_fmt(x, sf) for x in out) + (r"\ \mathrm{" + unit.replace(" ", r"\,") + "}" if unit else "")
    rows = [[v[0], (str(vals[k]).strip() or "?") + (" " + v[1] if v[1] else "")] for k, v in spec.items()]
    rows.append(["Rearranged", unk + " = " + str(sols[0])])
    return _res("", f[2], main, rows, "Answers to " + str(sf) + " significant figures.", _plain(out[0], sf))

def phys(op, a="", b="", c="", sf=3):
    try:
        if op == "list": return formula_list()
        if op == "convert": return op_convert(a, b, c, sf)
        if op == "suvat": return op_suvat(json.loads(a), int(sf))
        if op == "formula": return op_formula(a, json.loads(b), sf)
        raise ValueError("unknown physics operation")
    except (ValueError, KeyError, ZeroDivisionError, TypeError, sp.SympifyError, SyntaxError) as ex:
        return json.dumps({"kind": "bad", "badge": "Could not compute", "note": str(ex)[:160], "exact": r"\text{—}", "dec": r"\text{—}", "text": "", "rows": [], "chem": True})
